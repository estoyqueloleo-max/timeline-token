package legacy

import (
	"context"
	"encoding/json"
	"fmt"
	"io"
	"log"
	"net/http"
	"net/url"
	"os"
	"os/signal"
	"path/filepath"
	"syscall"

	"github.com/mark3labs/mcp-go/mcp"
	"github.com/mark3labs/mcp-go/server"
)

// SearxngResult representa un resultado individual de SearXNG
type SearxngResult struct {
	Title   string `json:"title"`
	URL     string `json:"url"`
	Content string `json:"content"`
}

// SearxngResponse representa la respuesta JSON de SearXNG
type SearxngResponse struct {
	Results []SearxngResult `json:"results"`
}

func main() {
	// Configurar logs a archivo
	ex, _ := os.Executable()
	cwd := filepath.Dir(ex)
	logPath := filepath.Join(cwd, "mcp-searxng.log")
	logFile, _ := os.OpenFile(logPath, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0666)
	if logFile != nil {
		multiWriter := io.MultiWriter(os.Stderr, logFile)
		log.SetOutput(multiWriter)
	}
	log.Printf("--- Iniciando mcp-searxng en: %s ---\n", logPath)

	// Crear Servidor MCP
	s := server.NewMCPServer("searxng-search", "1.0.0")

	// Herramienta de Búsqueda
	searchTool := mcp.Tool{
		Name:        "search",
		Description: "Busca información en internet usando SearXNG",
		InputSchema: mcp.ToolInputSchema{
			Type: "object",
			Properties: map[string]interface{}{
				"query": map[string]interface{}{
					"type":        "string",
					"description": "El término de búsqueda",
				},
			},
			Required: []string{"query"},
		},
	}

	s.AddTool(searchTool, func(ctx context.Context, request mcp.CallToolRequest) (*mcp.CallToolResult, error) {
		args, _ := request.Params.Arguments.(map[string]interface{})
		query, _ := args["query"].(string)

		log.Printf("[TOOL: search] Buscando: %s\n", query)

		// URL de SearXNG (usamos la IP proporcionada por el usuario)
		searxUrl := fmt.Sprintf("http://192.168.1.4:8080/search?q=%s&format=json", url.QueryEscape(query))

		client := &http.Client{}
		req, _ := http.NewRequest("GET", searxUrl, nil)
		// Añadimos headers de navegador real para evitar el 403 Forbidden
		req.Header.Set("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36")
		req.Header.Set("Accept", "application/json, text/javascript, */*; q=0.01")
		req.Header.Set("Accept-Language", "es-ES,es;q=0.9,en;q=0.8")
		req.Header.Set("Referer", "http://192.168.1.4:8080/")

		resp, err := client.Do(req)
		if err != nil {
			log.Printf("[TOOL: search] ERROR: %v\n", err)
			return mcp.NewToolResultError(fmt.Sprintf("Error conectando a SearXNG: %v", err)), nil
		}
		defer resp.Body.Close()

		bodyBytes, _ := io.ReadAll(resp.Body)
		if resp.StatusCode != http.StatusOK {
			log.Printf("[TOOL: search] ERROR STATUS: %d\n", resp.StatusCode)
			log.Printf("[TOOL: search] RESPONSE PREVIEW: %s\n", string(bodyBytes[:func() int {
				if len(bodyBytes) > 200 {
					return 200
				}
				return len(bodyBytes)
			}()]))
			return mcp.NewToolResultError(fmt.Sprintf("SearXNG respondió con error %d. Revisa la configuración de 'limiter' o 'formats' en settings.yml", resp.StatusCode)), nil
		}

		var searxResp SearxngResponse
		if err := json.Unmarshal(bodyBytes, &searxResp); err != nil {
			log.Printf("[TOOL: search] ERROR DECODING: %v\n", err)
			log.Printf("[TOOL: search] RESPONSE PREVIEW: %s\n", string(bodyBytes[:func() int {
				if len(bodyBytes) > 200 {
					return 200
				}
				return len(bodyBytes)
			}()]))
			return mcp.NewToolResultError("Error al procesar la respuesta de SearXNG (se recibió HTML en lugar de JSON). Revisa que 'json' esté en 'search.formats' de tu settings.yml."), nil
		}

		// Formateamos los primeros 5 resultados
		var output string
		for i, res := range searxResp.Results {
			if i >= 5 {
				break
			}
			output += fmt.Sprintf("### %s\nURL: %s\n%s\n\n", res.Title, res.URL, res.Content)
		}

		if output == "" {
			output = "No se encontraron resultados para: " + query
		}

		log.Printf("[TOOL: search] ÉXITO: %d resultados encontrados\n", len(searxResp.Results))
		return mcp.NewToolResultText(output), nil
	})

	// Manejo de señales para limpieza
	sigChan := make(chan os.Signal, 1)
	signal.Notify(sigChan, os.Interrupt, syscall.SIGTERM)
	go func() {
		sig := <-sigChan
		log.Printf("Recibida señal %v, cerrando mcp-searxng...\n", sig)
		os.Exit(0)
	}()

	// Iniciar servidor
	if err := server.ServeStdio(s); err != nil {
		fmt.Fprintf(os.Stderr, "Error: %v\n", err)
		os.Exit(1)
	}
}
